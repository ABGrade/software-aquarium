cmake_minimum_required(VERSION 3.20)
project(SoftwareAquarium LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

add_library(aquarium_core src/renderer.cpp src/simulation.cpp)
target_include_directories(aquarium_core PUBLIC inc)
target_link_libraries(aquarium_core PUBLIC Threads::Threads)
if(MINGW)
    add_link_options(-static -static-libgcc -static-libstdc++)
endif()
if(WIN32)
    add_executable(aquarium WIN32 src/main.cpp src/app.cpp)
    target_compile_definitions(aquarium PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(aquarium PRIVATE aquarium_core gdi32 user32)
    add_executable(aquarium_snapshot src/snapshot.cpp src/app.cpp)
    target_compile_definitions(aquarium_snapshot PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(aquarium_snapshot PRIVATE aquarium_core gdi32 user32)
    if(MINGW)
        target_link_options(aquarium PRIVATE -municode)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(core_checks tests/core_checks.cpp)
    target_link_libraries(core_checks PRIVATE aquarium_core)
    add_test(NAME rendering_and_physics COMMAND core_checks)
endif()
